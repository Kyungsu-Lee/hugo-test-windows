{{ define "main" }} {{- $s := .Site -}} {{- $sections := $s.Params.mainSections | default (slice "posts") -}} {{- $posts := where $s.RegularPages "Type" "in" $sections -}} {{- $n := int ($s.Params.postsOnHome | default 6) -}} {{- /* postsOnHomeMax renders further posts that a "Show more" button reveals in place — the middle ground between a short home page and the full archive, without a second request. Leave it unset (or at or below postsOnHome) and nothing changes. The extras are rendered visible and JavaScript hides them, not the other way round, so a reader without JavaScript gets more posts rather than a button that does nothing. */ -}} {{- $max := int ($s.Params.postsOnHomeMax | default $n) -}} {{- if lt $max $n }}{{ $max = $n }}{{ end -}} {{- $recent := first $max $posts -}} {{- /* Two independent switches, the same shape as theme × ground: the header and the post list vary separately, so four home pages come out of two params. Unknown values warn and fall back rather than rendering an empty page — a typo in hugo.toml should cost a warning, not the home. */ -}} {{- $headers := slice "masthead" "profile" -}} {{- $header := $s.Params.homeHeader | default "masthead" -}} {{- if not (in $headers $header) -}} {{- warnf "nuno: unknown params.homeHeader %q, falling back to \"masthead\" (one of: %s)" $header (delimit $headers ", ") -}} {{- $header = "masthead" -}} {{- end -}} {{- $styles := slice "grid" "ledger" -}} {{- $style := $s.Params.homeStyle | default "grid" -}} {{- if not (in $styles $style) -}} {{- warnf "nuno: unknown params.homeStyle %q, falling back to \"grid\" (one of: %s)" $style (delimit $styles ", ") -}} {{- $style = "grid" -}} {{- end -}} {{- /* Where "Archive · N" points. Prefer a real /archives/ page. */ -}} {{- $archive := "" -}} {{- with $s.GetPage "/archives" }}{{ $archive = .RelPermalink }}{{ else }} {{- with $s.GetPage (printf "/%s" (index $sections 0)) }}{{ $archive = .RelPermalink }}{{ end -}} {{- end -}} {{ partial (printf "home/header-%s.html" $header) . }}
