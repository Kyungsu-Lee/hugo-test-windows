{{- /* Giscus comments. Off unless configured, and even then the request is not made until the reader scrolls to it. This is the theme's only third-party connection, and it is the reader's bandwidth and the reader's data, so it is opt-in twice over: nothing loads without a params.giscus block, and nothing is fetched on a page view that never reaches the comments. All four ids are required. Giscus fails silently with a partial config — an empty box and no explanation — so an incomplete block warns loudly at build time and renders nothing rather than shipping that. `comments: false` turns them off — in front matter for one page, or in site params for the whole site. Takes the page as its argument. */ -}} {{- $s := .Site -}} {{- $g := $s.Params.giscus -}} {{- /* .Param, not .Params: it falls back from the page to the site, so a site wide `comments: false` turns them off everywhere and a page can still override. .Params would have read page front matter only, and a site-wide switch would have done nothing. */ -}} {{- $enabled := .Param "comments" -}} {{- if and $g (ne $enabled false) -}} {{- $missing := slice -}} {{- range (slice "repo" "repoId" "category" "categoryId") -}} {{- if not (index $g .) }}{{ $missing = $missing | append . }}{{ end -}} {{- end -}} {{- if $missing -}} {{- warnf "nuno: params.giscus is missing %s — comments not rendered. All four of repo, repoId, category, categoryId are required; find them at https://giscus.app" (delimit $missing ", ") -}} {{- else -}} {{- /* Giscus takes its own two-letter code; fall back to the site's. */ -}} {{- $lang := $g.lang | default (index (split $s.Language.Locale "-") 0) -}}
{{ i18n "comments" }}
{{- end -}} {{- end -}}
